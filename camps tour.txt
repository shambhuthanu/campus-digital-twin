# ---------------- DATABASE CONNECTION ----------------

def get_db_connection():
    connection = sqlite3.connect(DATABASE)
    connection.row_factory = sqlite3.Row
    return connection


# ---------------- CREATE DATABASE ----------------

def create_database():

    connection = get_db_connection()

    connection.execute("""
        CREATE TABLE IF NOT EXISTS events (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            date TEXT NOT NULL,
            venue TEXT NOT NULL,
            description TEXT
        )
    """)

    connection.execute("""
        CREATE TABLE IF NOT EXISTS resources (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            location TEXT NOT NULL,
            available INTEGER NOT NULL,
            total INTEGER NOT NULL
        )
    """)

    connection.execute("""
        CREATE TABLE IF NOT EXISTS complaints (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            student_name TEXT NOT NULL,
            location TEXT NOT NULL,
            description TEXT NOT NULL,
            category TEXT NOT NULL,
            status TEXT NOT NULL
        )
    """)

    # Check whether events already exist
    event_count = connection.execute(
        "SELECT COUNT(*) FROM events"
    ).fetchone()[0]

    if event_count == 0:

        # Event 1
        connection.execute("""

        ))

            VALUES (?, ?, ?, ?)
        """, (
            "College Cultural Fest",
            "2026-09-25",
            "Main Auditorium",
            "Annual cultural celebration for students."
        ))

    # Check whether resources already exist
    resource_count = connection.execute(
        "SELECT COUNT(*) FROM resources"

            INSERT INTO resources

        ))

        ))

        connection.execute("""
            INSERT INTO resources
            (name, location, available, total)
            VALUES (?, ?, ?, ?)
        """, (
            "Projectors",

# ---------------- SHORTEST PATH ----------------

def shortest_path(start, end):

    queue = [(0, start, [start])]
    visited = set()

    while queue:

        distance, current_place, path = heapq.heappop(queue)

        if current_place in visited:
            continue

        visited.add(current_place)

        if current_place == end:
            return path, distance

        for neighbour, cost in campus_graph[current_place].items():

            if neighbour not in visited:

                heapq.heappush(
                    queue,
                    (
                        distance + cost,
                        neighbour,
                        path + [neighbour]
                    )
                )

    return [], 0


# ---------------- COMPLAINT CLASSIFICATION ----------------

def classify_complaint(text):

    text = text.lower()

    if any(word in text for word in [
        "wifi",
        "internet",
        "network"
    ]):
        return "Network"

    if any(word in text for word in [
        "fan",
        "light",
        "electric",
        "power"
    ]):
        return "Electrical"

    if any(word in text for word in [
        "water",
        "leak",
        "washroom",
        "tap"
    ]):
        return "Plumbing"

    if any(word in text for word in [
        "computer",
        "projector",
        "keyboard"
    ]):
        return "Equipment"

    return "General Maintenance"


# ---------------- HOME PAGE ----------------

@app.route("/")
def home():

    connection = get_db_connection()

    events = connection.execute(
        "SELECT * FROM events ORDER BY date LIMIT 3"
    ).fetchall()

    resources = connection.execute(
        "SELECT * FROM resources"
    ).fetchall()

    complaints = connection.execute(
        "SELECT * FROM complaints ORDER BY id DESC"
    ).fetchall()

    connection.close()

    return render_template(
        "index.html",
        locations=campus_graph.keys(),
        events=events,
        resources=resources,
        complaints=complaints
    )


# ---------------- FIND ROUTE ----------------

@app.route("/find-route", methods=["POST"])
def find_route():

    start = request.form["start"]
    destination = request.form["destination"]

    path, distance = shortest_path(
        start,
        destination
    )

    connection = get_db_connection()

    events = connection.execute(
        "SELECT * FROM events ORDER BY date LIMIT 3"
    ).fetchall()

    resources = connection.execute(
        "SELECT * FROM resources"
    ).fetchall()

    complaints = connection.execute(
        "SELECT * FROM complaints ORDER BY id DESC"
    ).fetchall()

    connection.close()

    return render_template(
        "index.html",
        locations=campus_graph.keys(),
        events=events,
        resources=resources,
        complaints=complaints,
        route_path=" → ".join(path),
        route_distance=distance,
        walking_time=distance * 2
    )


# ---------------- COMPLAINT ----------------

@app.route("/complaint", methods=["POST"])
def complaint():

    student_name = request.form["student_name"]
    location = request.form["location"]
    description = request.form["description"]

    category = classify_complaint(description)

    connection = get_db_connection()

    connection.execute("""
        INSERT INTO complaints
        (student_name, location, description, category, status)
        VALUES (?, ?, ?, ?, ?)
    """, (
        student_name,
        location,
        description,
        category,
        "Open"
    ))

    connection.commit()
    connection.close()

    return redirect(url_for("home"))

# ---------------- ADMIN DASHBOARD ----------------

@app.route("/admin", methods=["GET", "POST"])
def admin():

    connection = get_db_connection()

    # Add new event
    if request.method == "POST":

        event_name = request.form["event_name"]
        event_date = request.form["event_date"]
        event_venue = request.form["event_venue"]
        event_description = request.form["event_description"]

        connection.execute("""
            INSERT INTO events
            (name, date, venue, description)
            VALUES (?, ?, ?, ?)
        """, (
            event_name,
            event_date,
            event_venue,
            event_description
        ))

        connection.commit()


    # Get all events
    events = connection.execute(
        "SELECT * FROM events ORDER BY date"
    ).fetchall()


    # Get all resources
    resources = connection.execute(
        "SELECT * FROM resources ORDER BY id"
    ).fetchall()


    # Get all complaints
    complaints = connection.execute(
        "SELECT * FROM complaints ORDER BY id DESC"
    ).fetchall()


    connection.close()


    return render_template(
        "admin.html",
        events=events,
        resources=resources,
        complaints=complaints
    )
